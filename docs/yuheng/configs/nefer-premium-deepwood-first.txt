# Stage 1 experimental config. Not a damage benchmark and not an optimized rotation.
# Team: Nefer / Lauma / Columbina / Sucrose. Easier Deepwood Sucrose setup.
# Source rotation, KQM Lauma Quick Guide, Nefer — Columbina — Sucrose section:
#   (Nefer E)¹ > Columbina E Q > Lauma (N1)¹ Q hE > Sucrose ED > Nefer combo > Sucrose ED
#   ¹ first rotation only.
# "Nefer combo" is translated as the Nefer guide's preferred dash-cancel string
# E 3[CD] E 3[CD]. gcsim has no bracket syntax, so that is written out.
# Sucrose's own guide writes the same opening but uses Sucrose E, not ED, on both
# casts. This file follows the Lauma guide's ED.
# No set_starting_verdant_dew and no other seeded combat resource.
# Energy starts full because that is the engine default. That is not a KQM energy calc.
# swap_delay=12 matches the other Yuheng validation files. It is not from KQM.
# One level-100 target. Single-target is an assumption.
# Talent 9 and the add-stats lines are assumptions, not a KQM artifact sheet.
# Weapons and sets are the KQM team recommendation:
#   Nefer 4pc Night of the Sky's Unveiling, R1 Reliquary of Truth.
#   Columbina 4pc Silken Moon's Serenade, R5 Prototype Amber (the guide's sustain option).
#   Lauma 2pc EM + 2pc ER, R1 Etherlight Spindlelute.
#   Sucrose 4pc Deepwood Memories, R5 Sacrificial Fragments.
# Gilded Dreams 2pc and Emblem 2pc are the stand-in for "2pc mixed EM/ER".
# Nefer, Lauma, and Columbina hitmarks used by these cancels are placeholders.

options iteration=3 duration=45 swap_delay=12;
target lvl=100 resist=0.1;

nefer char lvl=90/90 cons=0 talent=9,9,9;
nefer add weapon="reliquaryoftruth" refine=1 lvl=90/90;
nefer add set="nightoftheskysunveiling" count=4;
nefer add stats hp=4780 atk=311 em=374 cr=0.20 cd=0.622 er=0.10;

columbina char lvl=90/90 cons=0 talent=9,9,9;
columbina add weapon="prototypeamber" refine=5 lvl=90/90;
columbina add set="silkenmoonsserenade" count=4;
columbina add stats hp=4780 hp%=0.466 atk=311 cr=0.311 cd=0.20 er=0.518 em=80;

lauma char lvl=90/90 cons=0 talent=9,9,9;
lauma add weapon="etherlightspindlelute" refine=1 lvl=90/90;
lauma add set="gildeddreams" count=2;
lauma add set="emblemofseveredfate" count=2;
lauma add stats hp=4780 atk=311 em=561 er=0.30 cr=0.15 cd=0.20;

sucrose char lvl=90/90 cons=0 talent=6,6,6;
sucrose add weapon="sacrificialfragments" refine=5 lvl=90/90;
sucrose add set="deepwoodmemories" count=4;
sucrose add stats hp=4780 atk=311 em=561 er=0.20 cr=0.10 cd=0.15;

active nefer;

# First rotation only: Nefer E, then Lauma N1.
nefer skill;
columbina skill, burst;
lauma attack, burst, skill[hold=1];
sucrose skill, dash;
nefer skill, charge, dash, charge, dash, charge, dash, skill, charge, dash, charge, dash, charge, dash;
sucrose skill, dash;
